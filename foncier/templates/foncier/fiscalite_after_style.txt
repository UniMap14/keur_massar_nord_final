  <!-- ── CADRE GÉNÉRAL ── -->
  <div class="tab-panel active" id="panel-general">

    <div class="info-band a-animer">
      <i class="fa-solid fa-circle-info"></i>
      <p><strong>Système fiscal sénégalais :</strong> Le Sénégal dispose d'un Code Général des Impôts (CGI) codifiant l'ensemble des règles fiscales. La DGID est chargée de l'assiette, du contrôle et du recouvrement des impôts directs et indirects.</p>
    </div>

    <div class="intro-grid">
      <div class="intro-card a-animer">
        <div class="intro-card-icon"><i class="fa-solid fa-scale-balanced"></i></div>
        <h3>Principes fondamentaux</h3>
        <p>Le système fiscal sénégalais repose sur les principes d'équité, de progressivité et de légalité. Tout impôt doit être institué par la loi.</p>
        <ul>
          <li>Obligation déclarative annuelle pour tout contribuable</li>
          <li>Principe de l'annualité de l'impôt</li>
          <li>Droits et garanties du contribuable lors des contrôles</li>
          <li>Possibilité de recours administratif et juridictionnel</li>
        </ul>
      </div>
      <div class="intro-card a-animer">
        <div class="intro-card-icon"><i class="fa-solid fa-sitemap"></i></div>
        <h3>Organisation de la DGID</h3>
        <p>La Direction Générale est organisée autour de trois grandes directions opérationnelles :</p>
        <ul>
          <li><strong>Direction des Impôts :</strong> Fiscalité directe et indirecte</li>
          <li><strong>Direction du Domaine :</strong> Gestion du patrimoine de l'État</li>
          <li><strong>Direction du Cadastre :</strong> Topographie et cartographie foncière</li>
          <li><strong>Conservation Foncière :</strong> Immatriculation des immeubles</li>
        </ul>
      </div>
    </div>

    <h2 class="section-title a-animer">Les principaux impôts au Sénégal</h2>
    <p class="section-sub a-animer">Vue d'ensemble des taxes et impôts gérés par la DGID.</p>

    <div class="impots-grid">
      <div class="impot-card a-animer">
        <div class="impot-card-ic ic-or"><i class="fa-solid fa-user-tie"></i></div>
        <h4>Impôt sur le Revenu (IR)</h4>
        <p>Impôt progressif annuel sur les revenus des personnes physiques (salaires, revenus fonciers, BIC...).</p>
        <span class="taux"><i class="fa-solid fa-percent"></i> 0% à 40%</span>
      </div>
      <div class="impot-card a-animer">
        <div class="impot-card-ic ic-bl"><i class="fa-solid fa-building"></i></div>
        <h4>Impôt sur les Sociétés (IS)</h4>
        <p>Impôt sur les bénéfices des personnes morales (SA, SARL, GIE...) résidentes au Sénégal.</p>
        <span class="taux"><i class="fa-solid fa-percent"></i> 30%</span>
      </div>
      <div class="impot-card a-animer">
        <div class="impot-card-ic ic-gr"><i class="fa-solid fa-tag"></i></div>
        <h4>Taxe sur la Valeur Ajoutée (TVA)</h4>
        <p>Taxe sur la consommation collectée par les assujettis et reversée mensuellement à la DGID.</p>
        <span class="taux"><i class="fa-solid fa-percent"></i> 18%</span>
      </div>
      <div class="impot-card a-animer">
        <div class="impot-card-ic ic-pu"><i class="fa-solid fa-store"></i></div>
        <h4>Patente</h4>
        <p>Contribution annuelle des commerçants, industriels et prestataires de services exerçant au Sénégal.</p>
        <span class="taux"><i class="fa-solid fa-info-circle"></i> Variable / CA</span>
      </div>
      <div class="impot-card a-animer">
        <div class="impot-card-ic ic-re"><i class="fa-solid fa-house"></i></div>
        <h4>Contribution Foncière (CFPB)</h4>
        <p>Impôt annuel sur les propriétés bâties dû par tout propriétaire d'immeuble sur le territoire.</p>
        <span class="taux"><i class="fa-solid fa-percent"></i> 5% à 7,5%</span>
      </div>
      <div class="impot-card a-animer">
        <div class="impot-card-ic ic-te"><i class="fa-solid fa-briefcase"></i></div>
        <h4>Taxe Professionnelle Unique (TPU)</h4>
        <p>Régime simplifié pour les micro-entreprises dont le chiffre d'affaires est inférieur à 50 millions FCFA.</p>
        <span class="taux"><i class="fa-solid fa-percent"></i> 5% à 8% du CA</span>
      </div>
    </div>

    <h2 class="section-title">Vos obligations fiscales en tant que particulier</h2>
    <p class="section-sub">Découvrez les démarches à effectuer selon votre situation personnelle.</p>

    <div class="info-band">
      <i class="fa-solid fa-circle-info"></i>
      <p><strong>Première déclaration ?</strong> Commencez par obtenir votre NIF (Numéro d'Identification Fiscale) auprès du centre des impôts le plus proche ou en ligne via votre <a href="espace-contribuable.html" style="color:var(--green);font-weight:600;">espace contribuable</a>.</p>
    </div>

    <div class="intro-grid" style="margin-bottom:36px;">
      <div class="intro-card">
        <div class="intro-card-icon"><i class="fa-solid fa-file-invoice"></i></div>
        <h3>Revenus imposables</h3>
        <p>Sont soumis à l'IR :</p>
        <ul>
          <li>Traitements et salaires</li>
          <li>Revenus fonciers (loyers)</li>
          <li>Bénéfices industriels et commerciaux (BIC)</li>
          <li>Bénéfices non commerciaux (BNC)</li>
          <li>Revenus agricoles</li>
          <li>Revenus de capitaux mobiliers</li>
        </ul>
      </div>
      <div class="intro-card">
        <div class="intro-card-icon"><i class="fa-solid fa-shield-halved"></i></div>
        <h3>Exonérations et déductions</h3>
        <p>Bénéficiez de déductions pour :</p>
        <ul>
          <li>Charges de famille (abattement par enfant)</li>
          <li>Cotisations retraite et prévoyance</li>
          <li>Frais professionnels (forfait 20%)</li>
          <li>Intérêts d'emprunt immobilier</li>
          <li>Revenus &lt; 630 000 FCFA/an : exonérés</li>
        </ul>
      </div>
    </div>

    <h2 class="section-title">Démarche pas à pas</h2>
    <p class="section-sub">Comment déclarer votre impôt sur le revenu.</p>
    <div class="steps-list">
      <div class="step-item a-animer">
        <div class="step-num">1</div>
        <div class="step-body">
          <h4>Obtenez votre NIF</h4>
          <p>Munissez-vous de votre CNI et rendez-vous au centre des impôts ou créez votre compte en ligne sur l'<a href="espace-contribuable.html">espace contribuable</a>. Le NIF est délivré gratuitement sous 48h.</p>
        </div>
      </div>
      <div class="step-item a-animer">
        <div class="step-num">2</div>
        <div class="step-body">
          <h4>Rassemblez vos justificatifs</h4>
          <p>Fiches de paie, relevés bancaires, contrats de location, factures professionnelles... Conservez tous les documents de l'année écoulée.</p>
        </div>
      </div>
      <div class="step-item a-animer">
        <div class="step-num">3</div>
        <div class="step-body">
          <h4>Remplissez votre déclaration</h4>
          <p>Utilisez le formulaire DGID-IR-001 disponible sur la page <a href="formulaires.html">Formulaires</a> ou déclarez directement en ligne sur votre <a href="espace-contribuable.html">espace contribuable</a>.</p>
        </div>
      </div>
      <div class="step-item a-animer">
        <div class="step-num">4</div>
        <div class="step-body">
          <h4>Déposez avant le 30 avril</h4>
          <p>La déclaration annuelle d'IR doit être soumise avant le <strong>30 avril</strong> de l'année suivant l'exercice. Des pénalités de 25% s'appliquent en cas de retard.</p>
        </div>
      </div>
      <div class="step-item a-animer">
        <div class="step-num">5</div>
        <div class="step-body">
          <h4>Payez votre impôt</h4>
          <p>Réglez en ligne (carte, mobile money), au guichet ou par virement bancaire. Téléchargez votre quittance depuis votre espace personnel.</p>
        </div>
      </div>
    </div>

    <div class="warn-band">
      <i class="fa-solid fa-triangle-exclamation"></i>
      <p><strong>Rappel :</strong> Tout salarié dont l'employeur effectue déjà les retenues à la source doit quand même déposer une déclaration annuelle s'il dispose d'autres sources de revenus (loyers, activité secondaire...).</p>
    </div>

  </div><!-- /panel-particulier -->

  <div class="tab-panel" id="panel-procedures">

    <h2 class="section-title">Procédures fiscales</h2>
    <p class="section-sub">Guides pratiques pour les principales démarches auprès de la DGID.</p>

    <div class="proc-grid">

      <div class="proc-card a-animer">
        <div class="proc-card-head">
          <div class="proc-card-icon ic-gr"><i class="fa-solid fa-id-card"></i></div>
          <h3>Obtenir un NIF</h3>
        </div>
        <ul class="proc-steps">
          <li>Se munir d'une pièce d'identité valide (CNI ou passeport)</li>
          <li>Remplir le formulaire de demande de NIF</li>
          <li>Déposer au centre des impôts OU créer un compte en ligne</li>
          <li>Recevoir le NIF sous 48 heures ouvrables</li>
          <li>Le NIF est gratuit et valable à vie</li>
        </ul>
      </div>

      <div class="proc-card a-animer">
        <div class="proc-card-head">
          <div class="proc-card-icon ic-bl"><i class="fa-solid fa-file-circle-check"></i></div>
          <h3>Attestation de régularité</h3>
        </div>
        <ul class="proc-steps">
          <li>Être à jour dans vos obligations déclaratives</li>
          <li>Se connecter à l'espace contribuable</li>
          <li>Sélectionner "Demande d'attestation"</li>
          <li>Télécharger le document sous 24h</li>
          <li>Validité : 3 mois — Gratuit</li>
        </ul>
      </div>

      <div class="proc-card a-animer">
        <div class="proc-card-head">
          <div class="proc-card-icon ic-or"><i class="fa-solid fa-gavel"></i></div>
          <h3>Contester un redressement</h3>
        </div>
        <ul class="proc-steps">
          <li>Délai : 30 jours après notification</li>
          <li>Rédiger une réclamation motivée écrite</li>
          <li>Adresser au Directeur des Impôts compétent</li>
          <li>Si rejet : saisir la Commission de Conciliation</li>
          <li>Dernier recours : Tribunal régional hors classe</li>
        </ul>
      </div>

      <div class="proc-card a-animer">
        <div class="proc-card-head">
          <div class="proc-card-icon ic-pu"><i class="fa-solid fa-handshake"></i></div>
          <h3>Demande de facilités de paiement</h3>
        </div>
        <ul class="proc-steps">
          <li>Rédiger une demande motivée avant l'échéance</li>
          <li>Joindre les pièces justifiant la situation financière</li>
          <li>Adresser au Directeur des Impôts concerné</li>
          <li>Accord possible : paiement en 3 à 12 mensualités</li>
          <li>Des garanties peuvent être exigées</li>
        </ul>
      </div>

      <div class="proc-card a-animer">
        <div class="proc-card-head">
          <div class="proc-card-icon ic-te"><i class="fa-solid fa-search"></i></div>
          <h3>Vérification de comptabilité</h3>
        </div>
        <ul class="proc-steps">
          <li>Avis de vérification remis 8 jours avant</li>
          <li>Vous avez le droit d'être assisté d'un conseil</li>
          <li>Durée maximale : 6 mois (3 mois renouvelable)</li>
          <li>Notification des résultats par lettre recommandée</li>
          <li>30 jours pour répondre aux observations</li>
        </ul>
      </div>

      <div class="proc-card a-animer">
        <div class="proc-card-head">
          <div class="proc-card-icon ic-re"><i class="fa-solid fa-rotate-left"></i></div>
          <h3>Remboursement de crédit TVA</h3>
        </div>
        <ul class="proc-steps">
          <li>Remplir le formulaire DGID-RBT-006</li>
          <li>Joindre toutes les factures d'achats</li>
          <li>Déposer à la Direction des Impôts compétente</li>
          <li>Instruction du dossier : 3 mois maximum</li>
          <li>Remboursement par virement bancaire</li>
        </ul>
      </div>

    </div>

    <div class="info-band a-animer">
      <i class="fa-solid fa-circle-info"></i>
      <p>Téléchargez les formulaires nécessaires à ces procédures sur la page <a href="formulaires.html" style="color:var(--green);font-weight:600;">Formulaires à télécharger</a>. Consultez également les textes législatifs applicables sur la page <a href="textes-legislatifs.html" style="color:var(--green);font-weight:600;">Textes Législatifs</a>.</p>

/* ── Calcul ── */
function calculer(){
  const type    = document.getElementById('simType').value;
  const revenu  = parseFloat(document.getElementById('simRevenu').value) || 0;
  const charges = parseFloat(document.getElementById('simCharges').value) || 0;
  const enfants = parseInt(document.getElementById('simEnfants').value) || 0;

  if (revenu <= 0) { alert('Veuillez saisir un revenu valide.'); return; }

  let impot = 0, net = revenu, abatt = 0, deductEnfants = 0, libType = '';

  if (type === 'ir') {
    // Abattement frais pro 20% (plafonné à 900 000 FCFA)
    abatt = Math.min(revenu * 0.20, 900000);
    // Déduction enfants : 90 000 FCFA par enfant (max 4)
    deductEnfants = Math.min(enfants, 4) * 90000;
    net = Math.max(revenu - abatt - deductEnfants, 0);
    // Barème progressif
    if      (net <= 630000)     impot = 0;
    else if (net <= 1500000)    impot = (net - 630000) * 0.20;
    else if (net <= 4000000)    impot = 174000 + (net - 1500000) * 0.30;
    else if (net <= 8000000)    impot = 924000 + (net - 4000000) * 0.35;
    else                        impot = 2324000 + (net - 8000000) * 0.40;
    libType = 'Impôt sur le Revenu (IR)';
  }
  else if (type === 'is') {
    net = Math.max(revenu - charges, 0);
    impot = net > 0 ? Math.max(net * 0.30, 500000) : 0;
    abatt = charges;
    libType = 'Impôt sur les Sociétés (IS)';
  }
  else if (type === 'tva') {
    impot = revenu * 0.18;
    net = revenu;
    libType = 'TVA (18%)';
  }
  else if (type === 'tpu') {
    net = Math.max(revenu - charges, 0);
    const taux = net <= 10000000 ? 0.05 : net <= 25000000 ? 0.06 : 0.08;
    impot = net * taux;
    abatt = charges;
    libType = 'Taxe Professionnelle Unique (TPU)';
  }

  impot = Math.round(impot);
  const tauxEffectif = revenu > 0 ? ((impot / revenu) * 100).toFixed(1) : 0;
  const netApres = Math.max(revenu - impot, 0);

  const fmt = v => Math.round(v).toLocaleString('fr-FR') + ' FCFA';

  document.getElementById('res-revenu').textContent  = fmt(revenu);
  document.getElementById('res-net').textContent     = fmt(net);
  document.getElementById('res-impot').textContent   = fmt(impot);
  document.getElementById('res-taux').textContent    = tauxEffectif + '%';
  document.getElementById('res-net-apres').textContent = fmt(netApres);

  const rowAbatt   = document.getElementById('row-abattement');
  const rowEnfants = document.getElementById('row-enfants');

  if (type === 'ir') {
    rowAbatt.style.display   = '';
    rowEnfants.style.display = '';
    document.getElementById('res-abatt').textContent   = '- ' + fmt(abatt);
    document.getElementById('res-enfants').textContent = '- ' + fmt(deductEnfants);
  } else if (type === 'is' || type === 'tpu') {
    rowAbatt.style.display   = '';
    rowEnfants.style.display = 'none';
    document.getElementById('res-abatt').textContent = '- ' + fmt(abatt);
    rowAbatt.querySelector('.result-label').textContent = 'Charges déductibles';
  } else {
    rowAbatt.style.display   = 'none';
    rowEnfants.style.display = 'none';
  }

  // Barème visuel (IR seulement)
  const bv = document.getElementById('bareme-visual');
  const bb = document.getElementById('bareme-bars');
  if (type === 'ir') {
    bv.style.display = '';
    const tranches = [
      {label:'0% → 630 000', max:630000, taux:0},
      {label:'20% → 1 500 000', max:1500000, taux:20},
      {label:'30% → 4 000 000', max:4000000, taux:30},
      {label:'35% → 8 000 000', max:8000000, taux:35},
      {label:'40% au-delà',   max:Infinity, taux:40},
    ];
    bb.innerHTML = '';
    tranches.forEach(function(t, i){
      const fill = net >= t.max ? 100 : Math.max(0, Math.min(100, ((net - (i > 0 ? [0,630000,1500000,4000000,8000000][i] : 0)) / (t.max - [0,630000,1500000,4000000,8000000][i])) * 100));
      if (net < (i > 0 ? [0,630000,1500000,4000000,8000000][i] : 0)) return;
      bb.innerHTML += '<div class="bareme-bar-wrap"><div class="bareme-bar-label"><span>'+t.label+'</span><span>'+t.taux+'%</span></div><div class="bareme-bar-track"><div class="bareme-bar-fill" style="width:'+Math.min(fill,100)+'%"></div></div></div>';
    });
  } else {
    bv.style.display = 'none';
  }

  document.getElementById('simEmpty').style.display  = 'none';
  document.getElementById('simResult').style.display = 'block';
}
